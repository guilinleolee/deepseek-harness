/* 微博评论爬取 - 修复下载版 */ (function(){ console.clear(); console.log('%c='.repeat(70), 'color: #667eea'); console.log('%c微博评论爬取 - 修复下载版', 'color: #667eea; font-size: 18px; font-weight: bold'); console.log('%c='.repeat(70), 'color: #667eea');
