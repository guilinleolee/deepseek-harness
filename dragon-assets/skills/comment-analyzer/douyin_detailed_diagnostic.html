'...' : '')); // 检查特征 const hasTime = /\\d/.test(text); const hasChinese = /[\\u4e00-\\u9fa5]/.test(text); const hasNumber = /\\d+/.test(text); console.log(' 特征: ' + (hasTime ? '⏰时间 ' : '') + (hasChinese ? '🈶中文 ' : '') + (hasNumber ? '🔢数字 ' : '') ); // 尝试提取用户名和内容 const lines = text.split(/\\n/).filter(l => l.trim()); if (lines.length > 0) { console.log(' 行数: ' + lines.length); lines.slice(0, 3).forEach((line, idx) => { console.log(' [' + (idx + 1) + '] ' + line.substring(0, 50)); }); } console.log(''); } // 查找可能包含评论的元素 console.log('\\n%c=== 智能查找评论 ===', 'color: #64ffda; font-weight: bold\\n'); let commentCandidates = []; // 查找包含时间信息的元素 const allElements = document.querySelectorAll('*'); const timePatterns = [ /\\d+秒前/, /\\d+分钟前/, /\\d+小时前/, /\\d+天前/, /刚刚/, /昨天/, /\\d{1,2}-\\d{1,2}/, // 01-15 格式 /\\d{4}-\\d{1,2}-\\d{1,2}/ // 2025-01-15 格式 ]; for (let i = 0; i < Math.min(allElements.length, 5000); i++) { const el = allElements[i]; const text = el.textContent || ''; // 跳过太长或太短的 if (text.length < 5 || text.length > 300) continue; // 跳过明显的容器 if (el.children.length > 8) continue; // 检查是否包含时间 for (const pattern of timePatterns) { if (pattern.test(text)) { // 检查父元素 const parent = el.parentElement; if (parent) { const parentClass = parent.className || ''; const grandParent = parent.parentElement; commentCandidates.push({ element: el, text: text, parentClass: parentClass, parentTag: parent.tagName, grandParentClass: grandParent ?
