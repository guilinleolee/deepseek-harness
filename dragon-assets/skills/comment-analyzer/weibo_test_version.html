🚀 第二步：完整爬取脚本

如果诊断脚本显示一切正常，运行完整爬取脚本：

✅ 已复制！ /* 微评评论完整爬取脚本 - 改进版 */ (function(){ console.clear(); console.log('%c='.repeat(60), 'color: #667eea'); console.log('%c微博评论爬取工具', 'color: #667eea; font-size: 18px; font-weight: bold'); console.log('%c='.repeat(60), 'color: #667eea'); if (!window.location.href.includes('weibo.com')) { console.error('❌ 请在微博页面使用'); return; } window.ALL_COMMENTS = new Map(); window.SCROLL_STATS = { iterations: 0, lastCount: 0, noNewCount: 0, maxIterations: 500, stopAfterNoNew: 100 }; function parseComment(text) { const colonIndex = text.indexOf(':'); if (colonIndex === -1) return null; const username = text.substring(0, colonIndex).trim(); const timeMatch = text.match(/(\d{1,2}-\d{1,2}-\d{1,2})\s+(\d{1,2}:\d{2})/); if (!timeMatch) return null; const time = timeMatch[1] + ' ' + timeMatch[2]; const timeStartIndex = text.indexOf(timeMatch[0]); let content = text.substring(colonIndex + 1, timeStartIndex).trim(); content = content.replace(/@[\u4e00-\u9fa5a-zA-Z0-9_]+/g, '').trim(); content = content.replace(/^:+/, '').trim(); content = content.replace(/\s+/g, ' '); if (content.length < 2) return null; const likeMatch = text.match(/(\d+)$/); const likes = likeMatch ? parseInt(likeMatch[1]) : 0; const uniqueId = username + '_' + content + '_' + time; return { username, content, time, likes, uniqueId }; } function extractComments() { const list = document.querySelector('.wbpro-list'); if (!list) return 0; const nodes = list.children; for (let i = 0; i < nodes.length; i++) { const text = nodes[i].textContent || ''; const comment = parseComment(text); if (comment && !window.ALL_COMMENTS.has(comment.uniqueId)) { window.ALL_COMMENTS.set(comment.uniqueId, comment); } } return window.ALL_COMMENTS.size; } function smartScroll() { const list = document.querySelector('.wbpro-list'); if (list) list.scrollTop = list.scrollHeight; window.scrollTo(0, document.body.scrollHeight); const btns = Array.from(document.querySelectorAll('*')).filter(el => { const t = el.textContent || ''; return (t.includes('加载更多') || t.includes('展开查看') || t.includes('点击查看')) && el.offsetParent !== null; }); if (btns.length > 0) btns[0].click(); } async function scrape() { console.log('开始爬取...'); let initialCount = extractComments(); console.log('初始评论数: ' + initialCount); if (initialCount === 0) { console.error('%c没有找到评论！', 'color: #dc3545; font-weight: bold'); console.log('请检查：'); console.log('1. 是否已登录微博'); console.log('2. 页面是否完全加载'); console.log('3. 是否滚动到评论区域'); return []; } for (let i = 0; i < window.SCROLL_STATS.maxIterations; i++) { window.SCROLL_STATS.iterations = i + 1; const count = extractComments(); if (i % 10 === 0 || count > window.SCROLL_STATS.lastCount) { console.log('[' + (i+1) + '/' + window.SCROLL_STATS.maxIterations + '] 已加载: ' + count + ' 条' + (count > window.SCROLL_STATS.lastCount ? ' (+新)' : '')); } if (count >= 680) { console.log('%c达到目标!', 'color: #28a745; font-weight: bold'); break; } if (count === window.SCROLL_STATS.lastCount) { window.SCROLL_STATS.noNewCount++; if (window.SCROLL_STATS.noNewCount % 20 === 0) { console.log('连续 ' + window.SCROLL_STATS.noNewCount + ' 次无新评论'); } if (window.SCROLL_STATS.noNewCount >= window.SCROLL_STATS.stopAfterNoNew) { console.log('%c停止: 连续无新评论', 'color: #ffc107'); break; } } else { window.SCROLL_STATS.noNewCount = 0; window.SCROLL_STATS.lastCount = count; } smartScroll(); await new Promise(r => setTimeout(r, 1000)); } const arr = Array.from(window.ALL_COMMENTS.values()).sort((a,b) => b.likes - a.likes); console.log('%c='.repeat(60), 'color: #667eea'); console.log('%c完成! 总计: ' + arr.length + ' 条', 'color: #28a745; font-weight: bold'); console.log('%c='.repeat(60), 'color: #667eea'); return arr; } async function download(data) { const blob = new Blob([JSON.stringify(data,null,2)], {type:'application/json'}); const url = URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = 'weibo_comments_' + new Date().toISOString().replace(/[:.]/g,'-').slice(0,19) + '.json'; a.click(); URL.revokeObjectURL(url); } scrape().then(comments => { if (comments.length === 0) return; const data = { url: window.location.href, target_count: 680, actual_count: comments.length, completion_rate: (comments.length/680*100).toFixed(1) + '%', extracted_at: new Date().toISOString(), comments: comments }; download(data); console.log('%c文件已下载!', 'color: #28a745; font-weight: bold'); }).catch(e => console.error('错误:', e)); })();
