🎵 抖音评论选择器诊断工具

帮助找到抖音评论区域的正确CSS选择器

⚠️ 使用说明：
  1. 打开抖音视频页面（确保能看到评论）
  2. 按 F12 打开浏览器控制台
  3. 输入 allow pasting 然后按 Enter
  4. 复制下方诊断脚本
  5. 粘贴到控制台并按 Enter
  6. 查看输出结果，找到能匹配评论的选择器
✅ 已复制！

📜 诊断脚本代码

if (!window.location.href.includes('douyin.com')) { console.error('❌ 请在抖音页面使用'); return; } // 常见的抖音评论选择器 const selectors = [ // 抖音PC网页版 '[class*="comment"]', '[class*="Comment"]', '.comment-item', '.reply-item', '[data-e2e="comment-item"]', '[data-e2e="reply-item"]', // 抖音移动端H5 '[class*="CommentItem"]', '[class*="ReplyItem"]', '.comment-list-item', '.reply-list-item', // 通用容器 '.comment-container', '.reply-container', '[class*="list"] [class*="item"]', ]; console.log(`[${index + 1}] ${selector}`); console.log(` 找到: ${count} 个元素`); if (count > 0) { // 分析第一个元素的内容 const first = elements[0]; const text = first.textContent || ''; const className = first.className || ''; console.log(` 类名: ${className}`); console.log(` 内容预览: ${text.substring(0, 100)}...`); // 检查是否包含评论特征 const hasComment = text.includes('评论') || text.includes('回复') || /\d+天前|\d+小时前|\d+分钟前/.test(text); if (hasComment) { console.log(` %c✅ 可能包含评论`, 'color: #28a745; } catch (e) { console.log(` ❌ 错误: ${e.message}`); console.log(''); } }); // 查找包含评论文本的元素 console.log('\n[search] 查找包含评论特征的元素...\n'); const allElements = document.querySelectorAll('*'); let commentElements = []; if (parent) { const parentClass = parent.className || ''; const parentTag = parent.tagName.toLowerCase(); commentElements.push({ element: el, text: text.substring(0, 100), parentClass: parentClass, parentTag: parentTag }); } } if (commentElements.length >= 10) break; } if (commentElements.length > 0) { console.log(`%c找到 ${commentElements.length} 个可能的评论元素:`, 'color: #28a745; font-weight: bold'); commentElements.slice(0, 5).forEach((item, index) => { console.log(`\n[${index + 1}] 文本: ${item.text}...`); console.log(` 父元素: ${item.parentTag}.${item.parentClass}`); // 推荐选择器 if (item.parentClass) { const classSelector = '.' + item.parentClass.split(' ').join('.'); console.log(` 推荐选择器: ${classSelector}`); } }); } else { console.log('%c❌ 未找到评论元素', 'color: #dc3545; font-weight: bold'); } // 总结 console.log('\n' + '='.repeat(70)); console.log('%c诊断总结', 'color: #fe2c55; font-size: 18px; font-weight: bold'); console.log('='.repeat(70)); if (results.length > 0) { console.log(`\n✅ 找到 ${results.length} 个可能的选择器:`); results.forEach(r => { console.log(` • ${r.selector} (${r.count}个元素)`); }); console.log('\n推荐使用第一个选择器进行爬取。'); } else { console.log('\n⚠️ 未找到明确的选择器'); console.log('\n可能的原因:'); console.log(' 1. 页面还未加载完成，请稍后再试'); console.log(' 2. 需要登录才能查看评论'); console.log(' 3. 抖音更新了页面结构'); console.log('\n建议:'); console.log(' • 手动滚动页面让评论加载');
