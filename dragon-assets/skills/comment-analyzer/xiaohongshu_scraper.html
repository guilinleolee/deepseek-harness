'✓ 在小红书' : '✗ 不在小红书'); // 检查2: 查找可能的评论容器 console.log('\n[检查2] 查找评论容器...'); const possibleSelectors = [ '.comment-container', '.comment-list', '[class*="comment"]', '[class*="Comment"]', '.note-comments', '.comments-wrapper', '[class*="reply"]', '[class*="Reply"]' ]; 爬取脚本代码

/* 小红书评论爬取脚本 - 通用版 */ (function(){ console.clear(); console.log('%c='.repeat(70), 'color: #ff2442'); console.log('%c小红书评论爬取工具', 'color: #ff2442; font-size: 18px; font-weight: bold'); console.log('%c='.repeat(70), 'color: #ff2442'); if (!window.location.href.includes('xiaohongshu.com')) { console.error('❌ 请在小红书页面使用'); return; } window.XHS_COMMENTS = new Map(); window.XHS_STATS = { iterations: 0, lastCount: 0, noNewCount: 0, maxIterations: 300, loadMoreClicks: 0 }; // 通用评论解析器 function parseComment(element) { const text = element.textContent || ''; // 尝试提取用户名（通常在前面） let username = ''; const possibleName = text.split(/\s+/)[0]; if (possibleName && possibleName.length < 20) { username = possibleName; } // 尝试提取内容（去除用户名后的部分） let content = text.substring(username.length).trim(); // 尝试提取点赞数 let likes = 0; const likeMatch = text.match(/(\d+)\s*[赞赞]/); if (likeMatch) { likes = parseInt(likeMatch[1]); } // 尝试提取时间 let time = ''; const timeMatch = text.match(/(\d+[天月年小时分钟前]+)/); if (timeMatch) { time = timeMatch[1]; } // 生成唯一ID const uniqueId = username + '_' + content.substring(0, 20) + '_' + time; if (content.length < 2) return null; return { username: username || '未知用户', content: content, time: time, likes: likes, uniqueId: uniqueId }; } // 提取评论 - 尝试多种选择器 function extractComments() { const selectors = [ '.comment-container', '.comment-list', '[class*="comment-item"]', '[class*="CommentItem"]', '.note-comments', '[class*="reply-item"]' ]; let totalNew = 0; selectors.forEach(selector => { try { const containers = document.querySelectorAll(selector); containers.forEach(container => { const items = container.children; for (let i = 0; i < items.length; i++) { const comment = parseComment(items[i]); if (comment && !window.XHS_COMMENTS.has(comment.uniqueId)) { window.XHS_COMMENTS.set(comment.uniqueId, comment); totalNew++; } } }); } catch (e) {} }); return window.XHS_COMMENTS.size; } // 点击加载更多 function clickLoadMore() { const loadTexts = ['加载更多', '展开', '查看更多', '更多']; let clicked = false; const allElements = document.querySelectorAll('*'); for (const el of allElements) { const text = el.textContent || ''; if (text.length < 50) { for (const loadText of loadTexts) { if (text.includes(loadText) && el.offsetParent !== null) { const style = window.getComputedStyle(el); if (style.display !== 'none' && style.visibility !== 'hidden') { try { el.click(); clicked = true; window.XHS_STATS.loadMoreClicks++; console.log(`[按钮] 点击: "${text}"`); break; } catch (e) {} } } } if (clicked) break; } } return clicked; } // 主爬取函数 async function scrape() { console.log('\n[开始] 启动爬取...\n'); let initialCount = extractComments(); console.log(`[初始] 找到 ${initialCount} 条评论`); if (initialCount === 0) { console.error('%c❌ 没有找到评论！', 'color: #dc3545; font-size: 16px; font-weight: bold'); console.log('请确保：'); console.log('1. 已登录小红书'); console.log('2. 手动滚动到评论区'); console.log('3. 运行诊断脚本查看页面结构'); return []; } console.log(`\n[滚动] 开始加载评论...\n`); for (let i = 0; i < window.XHS_STATS.maxIterations; i++) { window.XHS_STATS.iterations = i + 1; const currentCount = extractComments(); if (i % 10 === 0 || currentCount > window.XHS_STATS.lastCount) { const newBadge = currentCount > window.XHS_STATS.lastCount ? ' (+新)' : ''; console.log(`[${i+1}/${window.XHS_STATS.maxIterations}] ${currentCount} 条${newBadge}`); } if (currentCount === window.XHS_STATS.lastCount) { window.XHS_STATS.noNewCount++; if (window.XHS_STATS.noNewCount >= 50) { console.log('%c连续无新评论，停止', 'color: #ffc107'); break; } } else { window.XHS_STATS.noNewCount = 0; window.XHS_STATS.lastCount = currentCount; } // 滚动并点击 window.scrollTo(0, document.body.scrollHeight); clickLoadMore(); await new Promise(r => setTimeout(r, 1500)); } const arr = Array.from(window.XHS_COMMENTS.values()); console.log('\n' + '='.repeat(70)); console.log(`%c[完成] 总计: ${arr.length} 条评论`, 'color: #28a745; font-size: 18px; font-weight: bold'); console.log('='.repeat(70) + '\n'); return arr; } // 下载 function download(data) { const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' }); const url = URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = 'xiaohongshu_comments_' + new Date().toISOString().replace(/[:.]/g, '-').slice(0, 19) + '.json'; a.click(); URL.revokeObjectURL(url); window.FINAL_DATA = data; } scrape().then(comments => { if (comments.length === 0) return; const data = { url: window.location.href, actual_count: comments.length, extracted_at: new Date().toISOString(), load_more_clicks: window.XHS_STATS.loadMoreClicks, comments: comments }; download(data); console.log('%c✅ 文件已下载！', 'color: #28a745; font-weight: bold'); console.log('%c手动复制: copy(JSON.stringify(window.FINAL_DATA, null, 2))', 'color: #ffc107'); }).catch(e => console.error('错误:', e)); })();

📖 使用帮助

🔑 关键点：
  • 必须登录：小红书需要登录才能查看评论
  • 手动滚动：先手动滚动触发评论加载
  • 先诊断：运行诊断脚本了解页面结构
  • 耐心等待：爬取需要10-20分钟

常见问题

Q: 找不到评论怎么办？

A: 先运行诊断脚本，把结果截图发给我，我会根据实际情况调整选择器。

Q: 只爬到几条评论？

A: 可能需要手动滚动几次触发加载，或者小红书有访问限制。

Q: 文件没有下载？

A: 在控制台输入 copy(JSON.stringify(window.FINAL_DATA, null, 2)) 然后粘贴到记事本保存。
