if (likesMatch) { likes = parseInt(likesMatch[1]); } // 清理内容 content = content.replace(/@[\u4e00-\u9fa5a-zA-Z0-9_]+/g, '').trim(); content = content.replace(/\s+/g, ' '); if (content.length < 1) return null; // 生成唯一ID const uniqueId = username + '_' + content.substring(0, 20) + '_' + timeStr; return { username: username || '匿名用户', content: content, time: timeStr, location: location, likes: likes, uniqueId: uniqueId }; } // 从DOM直接提取评论 function extractCommentsFromDOM() { const comments = []; const seen = new Set(); // 查找所有可能的评论元素 const selectors = [ '.comments-container > div', '[class*="comment-item"]', '[class*="comment"] > div', '.comment-container > div' ]; if (comments.length === 0) { console.error('%c❌ 没有找到评论！', 'color: #dc3545; font-size: 16px; font-weight: bold'); console.log('调试信息：尝试查找所有文本...'); // 输出调试信息 const allDivs = document.querySelectorAll('div'); count++; } } }); return []; } // 步骤3: 尝试滚动加载更多 console.log('\n[步骤3] 滚动加载更多...'); for (let i = 0; i < 15; i++) { window.scrollTo(0, document.body.scrollHeight); await new Promise(r => setTimeout(r, 1200)); } if (i % 5 === 0) { clickExpandButtons(); } } console.log('\n' + '='.repeat(70)); console.log(`%c[完成] 总计: ${comments.length} 条评论`, 'color: #28a745; font-size: 18px; font-weight: bold'); console.log('='.repeat(70) + '\n');
