📋 使用步骤

  1. 打开小红书笔记页面（确保能看到评论）
  2. 按键盘上的 F12 键打开开发者工具
  3. 点击顶部的 Console（控制台）标签
  4. 在控制台底部输入：allow pasting 然后按 Enter
  5. 点击下方"复制爬取脚本"按钮
  6. 在控制台按 Ctrl+V 粘贴脚本
  7. 按 Enter 执行脚本
  8. 等待 2-3 分钟，文件会自动下载
✅ 已复制到剪贴板！

📜 爬取脚本代码

/* 小红书评论爬取工具 v3.0 - 修复版 */ (function(){ console.clear(); console.log('%c='.repeat(70), 'color: #ff2442'); console.log('%c小红书评论爬取工具 v3.0', 'color: #ff2442; font-size: 18px; font-weight: bold'); console.log('%c='.repeat(70), 'color: #ff2442'); if (!window.location.href.includes('xiaohongshu.com')) { console.error('❌ 请在小红书页面使用'); return; } window.XHS_COMMENTS = []; window.XHS_STATS = { expandedCount: 0 }; // 从DOM元素解析单条评论 function parseCommentElement(element) { const text = element.textContent || ''; // 过滤无效内容 if (text.includes('共') && text.includes('条评论')) return null; if (text.includes('展开') || text.includes('收起')) return null; if (text.length < 10 || text.length > 500) return null; // 匹配时间 const timeMatch = text.match(/(昨天\s+\d{1,2}:\d{2})|(\d{1,2}-\d{1,2})|(\d+分钟前)|(\d+天前)|(\d+小时前)/); if (!timeMatch) return null; const timeStr = timeMatch[0]; const timeIndex = text.indexOf(timeStr); const beforeTime = text.substring(0, timeIndex).trim(); const afterTime = text.substring(timeIndex + timeStr.length).trim(); // 分离用户名和内容 let username = ''; let content = beforeTime; for (let i = 2; i <= Math.min(10, beforeTime.length); i++) { const possibleUser = beforeTime.substring(0, i); const possibleContent = beforeTime.substring(i).trim(); if (possibleContent.length >= 2 && possibleContent.length <= 200) { username = possibleUser; content = possibleContent; break; } } // 提取地点 let location = ''; const locationMatch = afterTime.match(/^([^\d赞]+?)([赞]|$)/); if (locationMatch && locationMatch[1].trim().length > 0) { location = locationMatch[1].trim(); } // 提取点赞数 let likes = 0; const likesMatch = afterTime.match(/(\d+)/); if (likesMatch) { likes = parseInt(likesMatch[1]); } // 清理内容 content = content.replace(/@[\u4e00-\u9fa5a-zA-Z0-9_]+/g, '').trim(); content = content.replace(/\s+/g, ' '); if (content.length < 1) return null; const uniqueId = username + '_' + content.substring(0, 20) + '_' + timeStr; return { username: username || '匿名用户', content: content, time: timeStr, location: location, likes: likes, uniqueId: uniqueId }; } // 从DOM提取所有评论 function extractCommentsFromDOM() { const comments = []; const seen = new Set(); const selectors = [ '.comments-container > div', '[class*="comment-item"]', '[class*="comment"] > div' ]; selectors.forEach(selector => { try { const elements = document.querySelectorAll(selector); elements.forEach(el => { if (el.children.length > 5) return; const comment = parseCommentElement(el); if (comment && !seen.has(comment.uniqueId)) { seen.add(comment.uniqueId); comments.push(comment); } }); } catch (e) {} }); return comments; } // 点击展开按钮 function clickExpandButtons() { let clicked = 0; const btns = document.querySelectorAll('.show-more, [class*="expand"]'); btns.forEach(btn => { const text = btn.textContent || ''; if ((text.includes('展开') || text.includes('更多')) && btn.offsetParent !== null) { try { btn.click(); clicked++; } catch (e) {} } }); return clicked; } // 主爬取函数 async function scrape() { console.log('\n[开始] 启动小红书评论爬取...\n'); // 展开回复 console.log('[步骤1] 展开所有回复...'); for (let i = 0; i < 3; i++) { const c = clickExpandButtons(); if (c > 0) console.log(`[展开] 点击了 ${c} 个按钮`); await new Promise(r => setTimeout(r, 800)); } // 提取评论 console.log('\n[步骤2] 提取评论...'); let comments = extractCommentsFromDOM(); console.log(`[初始] 提取到 ${comments.length} 条评论`); if (comments.length === 0) { console.error('%c❌ 没有找到评论！', 'color: #dc3545; font-size: 16px; font-weight: bold'); console.log('请确保：'); console.log('1. 已登录小红书'); console.log('2. 页面完全加载'); console.log('3. 能看到评论区域'); return []; } // 滚动加载更多 console.log('\n[步骤3] 滚动加载更多...'); for (let i = 0; i < 15; i++) { window.scrollTo(0, document.body.scrollHeight); await new Promise(r => setTimeout(r, 1200)); const newComments = extractCommentsFromDOM(); if (newComments.length > comments.length) { console.log(`[滚动] ${comments.length} → ${newComments.length} 条 (+${newComments.length - comments.length})`); comments = newComments; } if (i % 5 === 0) clickExpandButtons(); } console.log('\n' + '='.repeat(70)); console.log(`%c[完成] 总计: ${comments.length} 条评论`, 'color: #28a745; font-size: 18px; font-weight: bold'); console.log('='.repeat(70) + '\n'); // 预览前3条 console.log('%c前3条评论预览:', 'color: #64ffda'); comments.slice(0, 3).forEach((c, i) => { console.log(`\n${i + 1}. ${c.username}: ${c.content}`); console.log(` 📅 ${c.time} | 📍 ${c.location || '未知'} | 👍 ${c.likes}`); }); return comments; } // 下载文件 function download(data) { const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' }); const url = URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = 'xiaohongshu_comments_' + new Date().toISOString().replace(/[:.]/g, '-').slice(0, 19) + '.json'; a.click(); URL.revokeObjectURL(url); window.FINAL_DATA = data; } // 执行 scrape().then(comments => { if (comments.length === 0) return; const data = { url: window.location.href, actual_count: comments.length, extracted_at: new Date().toISOString(), expanded_replies: window.XHS_STATS.expandedCount, comments: comments }; download(data); console.log('\n%c✅ 文件已下载到 E:\\下载', 'color: #28a745; font-weight: bold'); console.log('%c如需手动复制: copy(JSON.stringify(window.FINAL_DATA, null, 2))', 'color: #ffc107'); }).catch(e => console.error('错误:', e)); })();

❓ 常见问题

Q: 只爬到几条评论怎么办？

A: 请确保先手动滚动一下评论区，让所有评论都加载完成。

Q: 提示"没有找到评论"？

A: 请确认已登录小红书，并且页面完全加载完成。

Q: 文件没有下载？

A: 使用手动复制方法，在控制台运行 copy(JSON.stringify(window.FINAL_DATA, null, 2))
