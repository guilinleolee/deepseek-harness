## ⭐ Solo Founder跑道守护 (V9.02新增)

> **化学催化剂原则应用于MVP层** — 识别PMF前的催化剂与抑制剂，让PMF信号自然浮现而非强行推动。

### MVP层化学催化剂原则

| 类型 | 定义 | 示例 |
|------|------|------|
| **Catalyst 催化剂** | 功能加速PMF信号浮现 | 真实付费用户反馈、推荐增长、复购行为 |
| **Inhibitor 抑制剂** | 功能延迟PMF信号浮现 | 装饰性UI、华丽动画、非核心文档 |
| **Toxic 有毒** | 功能增加认知负担 | 过早抽象、多余配置项、过度工程化 |

### PMF信号五级评估

| 级别 | 信号 | 描述 |
|------|------|------|
| 🔴 **无法忍受** | 用户主动询问在哪能买到 | 已有付费意愿但无购买渠道 |
| 🟠 **苦苦寻找** | 用户绕开障碍也要使用 | 痛点足够强，愿意容忍缺陷 |
| 🟡 **使用了并满意** | 用户主动推荐他人 | NPS≥40，有口碑增长 |
| 🟢 **有点满意** | 用户留存且活跃 | DAU稳定，有持续使用场景 |
| ⚪ **不确定** | 用户无明显反应 | 需进一步验证价值假设 |

### MVP Scope Limiter规则

在写代码前必须回答：

```
1. 这个功能是否让用户更接近PMF信号？

   - 如果YES → 继续2
2. 这个功能能否在3天内验证PMF假设？
   - 如果NO → 缩小范围
   - 如果YES → 继续3
3. 如果PMF假设错误，这个功能能快速废弃吗？
   - 如果NO → 设计成可丢弃
   - 如果YES → 实施
```

## 模块元素思维

### 代码元素分类

| 元素类型 | 特征 | 示例 | 管理方式 |
|----------|------|------|----------|
| **反应物（输入）** | 被处理 | 请求参数、文件 | 验证纯度 |
| **催化剂（工具）** | 促进反应 | Logger, Config | 注入使用 |
| **产物（输出）** | 产生结果 | Response, Event | 契约定义 |
| **副产物（副作用）** | 额外影响 | DB写入, API调用 | 显式记录 |
| **废物（错误）** | 失败情况 | Exception | 边界处理 |

### 副作用显式化

```typescript

    this.logger.log('Creating user'); // 隐藏的I/O

  }

class UserService {

    private db: Database,

    private eventBus: IEventBus,  // 显式副作用
  ) {}

  async createUser(data: UserData): Promise<UserCreated> {
    this.logger.info('Creating user', { email: data.email });
    const user = await this.db.users.create(data);
    this.eventBus.publish(new UserCreatedEvent(user)); // 显式副作用
    return user;
  }
}
```

### Stripe订阅集成

```typescript
// app/api/webhooks/stripe/route.ts
import { headers } from 'next/headers';
import Stripe from 'stripe';
import { db } from '@/lib/db';
import { subscriptions } from '@/lib/db/schema';

const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!);

export async function POST(req: Request) {
  const body = await req.text();
  const signature = headers().get('Stripe-Signature')!;

  let event: Stripe.Event;
  try {
    event = stripe.webhooks.constructEvent(
      body, signature, process.env.STRIPE_WEBHOOK_SECRET!
    );
  } catch (err) {
    return new Response('Webhook Error', { status: 400 });
  }

  switch (event.type) {
    case 'customer.subscription.created':
    case 'customer.subscription.updated': {
      const sub = event.data.object as Stripe.Subscription;
      await db.update(subscriptions)
        .set({
          status: sub.status,
          currentPeriodEnd: new Date(sub.current_period_end * 1000),
        })
        .where(eq(subscriptions.stripeCustomerId, sub.customer));
      break;
    }
    case 'customer.subscription.deleted': {
      const sub = event.data.object as Stripe.Subscription;
      await db.update(subscriptions)
        .set({ status: 'canceled' })
        .where(eq(subscriptions.stripeCustomerId, sub.customer));
      break;
    }
  }

  return new Response(null, { status: 200 });
}
```

### Python Pattern: 异步上下文管理

```python
import asyncio
from contextlib import asynccontextmanager
from typing import AsyncGenerator

class DatabasePool:
    def __init__(self, max_connections: int = 10):
        self.max_connections = max_connections
        self._semaphore = asyncio.Semaphore(max_connections)

    @asynccontextmanager
    async def acquire(self) -> AsyncGenerator["Connection", None]:
        """异步上下文管理器，自动获取和释放连接"""
        async with self._semaphore:
            conn = await self._create_connection()
            try:
                yield conn
            finally:
                await conn.close()

# 使用
async def get_user(db: DatabasePool, user_id: str) -> User | None:
    async with db.acquire() as conn:
        row = await conn.fetchone(
            "SELECT * FROM users WHERE id = $1", user_id
        )
        return User(**row) if row else None
```

### Python Pattern: 工厂函数 + 类型注解

```python
from dataclasses import dataclass, field
from enum import Enum
from typing import Protocol

class OrderStatus(Enum):
    PENDING = "pending"
    CONFIRMED = "confirmed"
    SHIPPED = "shipped"
    DELIVERED = "delivered"

@dataclass
class OrderLine:
    product_id: str
    quantity: int
    unit_price: float

    @property
    def subtotal(self) -> float:
        return self.quantity * self.unit_price

@dataclass
class Order:
    id: str
    customer_id: str
    lines: list[OrderLine] = field(default_factory=list)
    status: OrderStatus = OrderStatus.PENDING

    @property
    def total(self) -> float:
        return sum(line.subtotal for line in self.lines)

    def add_line(self, line: OrderLine) -> None:
        self.lines.append(line)

class OrderFactory:
    @staticmethod
    def create(customer_id: str, *line_data) -> Order:
        order = Order(id=uuid4(), customer_id=customer_id)
        for product_id, qty, price in line_data:
            order.add_line(OrderLine(product_id, qty, price))
```

```python
import subprocess
from pathlib import Path
from typing import Optional

class BackendWrapper:
    def __init__(self, binary_path: str):
        self.binary = binary_path

    def execute(self, args: list[str], timeout: int = 30) -> dict:
        result = subprocess.run(
            [self.binary, *args],
            capture_output=True,
            text=True,
            timeout=timeout
        )
        return {
            'stdout': result.stdout,
            'stderr': result.stderr,
            'returncode': result.returncode,
            'success': result.returncode == 0

        )
```
